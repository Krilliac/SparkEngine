cmake_minimum_required(VERSION 3.25)
if(NOT DEFINED SPARK_ENGINE_VERSION OR SPARK_ENGINE_VERSION STREQUAL "")
    include("${CMAKE_CURRENT_LIST_DIR}/../cmake/SparkWindowsVersionInfo.cmake")
    spark_resolve_engine_version(SPARK_ENGINE_VERSION)
endif()
project(SparkCooker VERSION ${SPARK_ENGINE_VERSION} LANGUAGES CXX)
if(BUILD_TESTS)
    enable_testing()
endif()
if(NOT TARGET SparkAssetPipelineCore)
    add_subdirectory("../SparkAssetPipelineCore" "${CMAKE_CURRENT_BINARY_DIR}/SparkAssetPipelineCore")
endif()
add_executable(SparkCooker src/main.cpp)
target_compile_definitions(SparkCooker PRIVATE
    SPARK_COOKER_VERSION=\"${PROJECT_VERSION}\"
)
find_package(Threads REQUIRED)
target_link_libraries(SparkCooker PRIVATE SparkAssetPipelineCore Threads::Threads)
target_compile_features(SparkCooker PRIVATE cxx_std_23)
set_target_properties(SparkCooker PROPERTIES FOLDER "Tools")
install(TARGETS SparkCooker RUNTIME DESTINATION bin COMPONENT tools)
